cmake_minimum_required(VERSION 3.20)
project(Seam_Carving_Project VERSION 1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Ensure CMake uses vcpkg toolchain
# (Visual Studio will auto-pass this if you opened the CMake project via VS)
# set(CMAKE_TOOLCHAIN_FILE "C:/vcpkg/scripts/buildsystems/vcpkg.cmake" CACHE STRING "Vcpkg toolchain")

# Find all required packages
find_package(OpenCV REQUIRED core imgproc highgui imgcodecs)
find_package(glfw3 REQUIRED)
find_package(imgui REQUIRED)

# Create executable
add_executable(Seam_Carving_Project main.cpp "seam_carving_bonus.h" "seam_carving_bonus.cpp")

if (MSVC)
    target_compile_options(Seam_Carving_Project PRIVATE /utf-8)
endif()

# Link all libraries
target_link_libraries(Seam_Carving_Project PRIVATE
    ${OpenCV_LIBS}
    glfw
    opengl32
    imgui::imgui
)

# Include directories
target_include_directories(Seam_Carving_Project PRIVATE
    ${OpenCV_INCLUDE_DIRS}
)

message(STATUS "OpenCV include dirs: ${OpenCV_INCLUDE_DIRS}")
message(STATUS "OpenCV libs: ${OpenCV_LIBS}")
message(STATUS "GLFW found: ${glfw3_FOUND}")
message(STATUS "ImGui found: ${imgui_FOUND}")
